// Add to site php file.
function send_webhook_on_new_post($ID, $post) {
    // Target URL where you want to send the RSS/Post data
    $webhook_url = 'https://your-endpoint.com';

    // Structure the payload data
    $payload = array(
        'title'     => get_the_title($ID),
        'permalink' => get_permalink($ID),
        'author'    => get_the_author_meta('display_name', $post->post_author),
        'excerpt'   => get_the_excerpt($ID),
        'date'      => $post->post_date
    );

    // Send the remote POST request to your webhook
    wp_remote_post($webhook_url, array(
        'method'    => 'POST',
        'headers'   => array('Content-Type' => 'application/json; charset=utf-8'),
        'body'      => json_encode($payload),
        'timeout'   => 5,
    ));
}
// Fires automatically whenever a post status changes to "published"
add_action('publish_post', 'send_webhook_on_new_post', 10, 2);
